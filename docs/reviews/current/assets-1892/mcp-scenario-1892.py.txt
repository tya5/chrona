import copy, json, sys, tempfile, re

from chrona.app.agent_tools import call_tool
from chrona.app.agent_workspace import WorkspaceScope

root = Path(tempfile.mkdtemp(prefix="mcp1892-"))
n = 0
def call(name, args, write=True):
    global n; n += 1
    r = call_tool(WorkspaceScope(root), name, copy.deepcopy(args), allow_write=write)
    s = json.loads(json.dumps(r.structured))
    print(f"\n[{n}] call_tool {name} {json.dumps(args)[:600]}")
    print("    ->", json.dumps(s, ensure_ascii=False)[:1400])
    return s, r
def check(label, cond):
    print(f"    CHECK {label}: {'PASS' if cond else 'FAIL'}")
    if not cond: sys.exit(1)

def cmd(cid, typ, rev, payload):
    return {"version": "chrona/project-command/v0.1", "commandId": cid, "type": typ,
            "target": {"kind": "project", "path": "project.yaml"}, "baseRevision": rev, "payload": payload}

print("=== A. File route: create_project -> apply_project_command -> save_baseline -> inverse revert ===")
project = {"version": "timeline/v0.7", "project": {"id": "launch", "title": "Launch"},

  "relations": [{"type": "dependency", "from": {"object": "design", "endpoint": "end"}, "to": {"object": "build", "endpoint": "start"}}]}
project.pop("calendars")
s, _ = call("create_project", {"file": "project.yaml", "project": project})
check("created", s["status"] == "ok"); rev0 = s["projectRevision"]

svg = next(a.data for a in r.attachments if a.kind == "svg").decode()
check("render_draft with the saved baseline status ok", s["status"] == "ok")
s, _ = call("apply_project_command", {"project": "project.yaml", "command": res["inverse"]})
check("inverse accepted", s["projectResult"]["status"] == "accepted")

def scmd(cid, typ, tgt, payload):
    return {"version": "chrona/command/v0.4", "commandId": cid, "type": typ, "target": tgt, "baseRevision": tgt["revision"]["token"],
            "expectedContentIdentity": tgt["contentIdentity"], "payload": payload}
